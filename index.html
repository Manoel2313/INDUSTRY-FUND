import * as THREE from './lib/three.module.js'; import { createMap } from './map.js'; import { Player } from './player.js'; let scene, camera, renderer, clock; let player; let isGameActive = false; const canvas = document.getElementById('game-canvas'); const overlay = document.getElementById('overlay'); function init() { try { scene = new THREE.Scene(); scene.background = new THREE.Color(0x050510); scene.fog = new THREE.FogExp2(0x050510, 0.05); camera = new THREE.PerspectiveCamera(75, window.innerWidth / window.innerHeight, 0.1, 1000); camera.position.set(0, 1.6, 5); renderer = new THREE.WebGLRenderer({ canvas: canvas, antialias: true }); renderer.setSize(window.innerWidth, window.innerHeight); renderer.setPixelRatio(window.devicePixelRatio); renderer.shadowMap.enabled = true; renderer.toneMapping = THREE.ACESFilmicToneMapping; renderer.toneMappingExposure = 1.0; clock = new THREE.Clock(); setupLights(); createMap(scene); player = new Player(camera, scene); window.addEventListener('resize', onWindowResize); setupPointerLock(); // Loop de renderização imediato (mesmo sem clique) animate(); console.log("INDUSTRY-FUND inicializado com sucesso."); } catch (error) { console.error("Erro na inicialização:", error); overlay.innerHTML = "ERRO AO CARREGAR O JOGO.
Abra o console (F12) para ver o erro."; } } function setupLights() { const ambientLight = new THREE.AmbientLight(0x111122, 1.5); scene.add(ambientLight); const dirLight = new THREE.DirectionalLight(0x4466cc, 1.0); dirLight.position.set(5, 10, 7); dirLight.castShadow = true; scene.add(dirLight); const redLight = new THREE.PointLight(0xff0000, 2, 10); redLight.position.set(0, 3, -5); scene.add(redLight); } function setupPointerLock() { overlay.addEventListener('click', () => { // Tenta travar o mouse canvas.requestPointerLock = canvas.requestPointerLock || canvas.mozRequestPointerLock; const promise = canvas.requestPointerLock(); // Fallback caso o navegador bloqueie o Pointer Lock (Segurança) if (promise && promise.catch) { promise.catch((err) => { console.warn("Pointer Lock bloqueado pelo navegador. Jogando sem travar o mouse.", err); // Se falhar, apenas ativa o jogo mesmo assim (para debug) isGameActive = true; overlay.style.display = 'none'; }); } }); document.addEventListener('pointerlockchange', lockChangeAlert, false); document.addEventListener('mozpointerlockchange', lockChangeAlert, false); } function lockChangeAlert() { if (document.pointerLockElement === canvas || document.mozPointerLockElement === canvas) { isGameActive = true; overlay.style.display = 'none'; clock.getDelta(); } else { // Se o jogo foi pausado e ainda não tem erro, mostra o overlay de volta if (document.pointerLockElement !== null) { isGameActive = false; overlay.style.display = 'flex'; overlay.innerHTML = "JOGO PAUSADO
Clique para continuar"; } } } function onWindowResize() { camera.aspect = window.innerWidth / window.innerHeight; camera.updateProjectionMatrix(); renderer.setSize(window.innerWidth, window.innerHeight); } function animate() { requestAnimationFrame(animate); const delta = Math.min(clock.getDelta(), 0.1); if (isGameActive && player) { player.update(delta); } renderer.render(scene, camera); } init();
